\section{Introdução}

\begin{frame}{Contexto e problema\notexto{Cap. 1}}
  \begin{itemize}
    \item Autômatos finitos são um pilar da Ciência da Computação, mas a disciplina é
          \textbf{reconhecidamente difícil}.
    \item Morazán e Antunez (2014): o estudante projeta autômatos \textbf{sem poder experimentar}
          nem receber \textbf{retroalimentação imediata}, ao contrário da programação.
    \item As ferramentas mais difundidas (\textbf{JFLAP}) são \textbf{só visuais}:
          sem descrição textual, sem versionamento, \textbf{sem diagnóstico} de erros.
    \item A alternativa textual (\textbf{FSM}, em Racket) descreve autômatos como código,
          mas \textbf{não os desenha nem anima}.
  \end{itemize}
  \vspace{4pt}
  \begin{block}{Lacuna}
    Unir a \textbf{descrição textual}, o \textbf{diagnóstico de um compilador} e a
    \textbf{simulação visual} em uma única ferramenta.
  \end{block}
\end{frame}

\begin{frame}{Pergunta de pesquisa e proposta\notexto{Cap. 1}}
  \begin{block}{Pergunta de pesquisa}
    Como uma linguagem textual de domínio específico, processada por um compilador com
    diagnósticos localizados e acoplada a uma simulação visual, pode oferecer ao estudante
    \textbf{retroalimentação imediata} na construção e na execução de autômatos finitos?
  \end{block}
  \vspace{6pt}
  \centering
  \begin{tikzpicture}[
      node distance=9mm,
      parte/.style={draw=acento,fill=acento!10,rounded corners,align=center,
                    minimum height=13mm,text width=33mm,inner sep=4pt,font=\small},

    \node[parte,right=of c] (i) {\textbf{Interface gráfica}\\diagrama e simulação\\passo a passo};
    \draw[seta] (d)--(c); \draw[seta] (c)--(i);
    \node[below=3mm of c,font=\small] {\textbf{Autosim}: o estudante escreve o texto; o diagrama é gerado e animado automaticamente};
  \end{tikzpicture}
\end{frame}

\begin{frame}{Objetivos\notexto{Cap. 1}}
  \textbf{Objetivo geral:} desenvolver uma ferramenta educacional que integra a descrição textual
  de autômatos, sua compilação pedagógica e sua simulação visual em um único ambiente.
  \vspace{6pt}
  \begin{block}{Objetivos específicos}
    \begin{enumerate}
      \item Projetar uma \textbf{DSL} textual para AFDs e AFNs, com gramática em EBNF;
      \item Implementar um \textbf{compilador pedagógico} com diagnósticos localizados;
      \item Desenvolver um \textbf{simulador} de cadeias sobre os autômatos descritos;
      \item Construir uma \textbf{interface gráfica} que desenhe e anime o autômato;
      \item \textbf{Validar} a ferramenta com testes, autômatos clássicos e cenários de erro.
    \end{enumerate}
  \end{block}
  \vspace{2pt}
  {\footnotesize Validação \textbf{funcional}; a eficácia pedagógica com estudantes é trabalho futuro.}
\end{frame}

\section{Fundamentação Teórica}

\begin{frame}{Autômatos finitos\notexto{Seção 2.1}}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      Um \textbf{AFD} é uma quíntupla $M = (Q, \Sigma, \delta, q_0, F)$:
      \begin{itemize}\small
        \item $Q$: estados \quad $\Sigma$: alfabeto
        \item $\delta: Q \times \Sigma \to Q$: transições
        \item $q_0$: inicial \quad $F \subseteq Q$: finais
      \end{itemize}
      \vspace{4pt}
      No \textbf{AFN}, $\delta: Q \times (\Sigma \cup \{\varepsilon\}) \to \mathcal{P}(Q)$:
      vários destinos e transições vazias $\varepsilon$.
      \vspace{4pt}

      AFD e AFN são \textbf{equivalentes} (construção de subconjuntos) e reconhecem as
      \textbf{linguagens regulares}.
    \end{column}
    \begin{column}{0.53\textwidth}
      \centering
      \vspace{14pt}
      \begin{tikzpicture}[shorten >=1pt,node distance=2.3cm,on grid,>={Stealth[]},
          every state/.style={draw=acento,thick,fill=acento!8,minimum size=9mm}]
        \node[state,initial,initial text={}] (q0) {$q_0$};
        \node[state] (q1) [right=of q0] {$q_1$};
        \node[state,accepting] (q2) [right=of q1] {$q_2$};
        \path[->]
          (q0) edge[loop above] node{$b$} (q0)
          (q0) edge node[above]{$a$} (q1)
          (q1) edge[loop above] node{$a$} (q1)
          (q1) edge node[above]{$b$} (q2)
          (q2) edge[loop above] node{$a,b$} (q2);
      \end{tikzpicture}
      \vspace{8pt}

      \small Cadeias que \textbf{contêm ``$ab$''}\\[3pt]
      \textcolor{aceita}{\textbf{aceita:}} \texttt{ab}, \texttt{aab}, \texttt{bab}\\
      \textcolor{rejeita}{\textbf{rejeita:}} $\varepsilon$, \texttt{ba}, \texttt{bbb}\\[3pt]
      \footnotesize\emph{Reaparece na DSL e na interface.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Compiladores e linguagens de domínio específico\notexto{Seção 2.2}}
  \begin{center}
    \begin{tikzpicture}[
        node distance=6mm and 10mm,
        etapa/.style={draw=acento,fill=acento!10,rounded corners,align=center,

      \node[etapa,right=of b] (c) {Análise\\\textbf{Semântica}};
      \draw[seta] (a)--(b); \draw[seta] (b)--(c);
      \node[below=1mm of a,font=\footnotesize\itshape] {texto $\to$ \emph{tokens}};
      \node[below=1mm of b,font=\footnotesize\itshape] {\emph{tokens} $\to$ AST};
      \node[below=1mm of c,font=\footnotesize\itshape,text width=30mm,align=center] {referências e consistência};
    \end{tikzpicture}
  \end{center}
  \vspace{2pt}
  \begin{block}{DSL (Fowler, 2010)}
    Linguagem de \textbf{expressividade limitada}, voltada a um \textbf{domínio específico}.
    \begin{itemize}
      \item \textbf{Interna:} embutida em uma linguagem hospedeira (ex.: FSM, em Racket).
      \item \textbf{Externa:} sintaxe própria e \textbf{compilador dedicado} --- é o caso da \textbf{Autosim}.
    \end{itemize}
  \end{block}
\end{frame}

\section{Trabalhos Relacionados}

\begin{frame}{Panorama: quatro ferramentas gráficas e uma textual\notexto{Seções 3.1--3.5}}
  \begin{itemize}
    \item \textbf{JFLAP} --- referência mundial; cobre toda a hierarquia de Chomsky;
          construção \textbf{só gráfica}, sem descrição textual nem diagnóstico.
    \item \textbf{GAM} (UFLA) --- \emph{desktop}; limitado a 10 estados e 3 símbolos.
    \item \textbf{Simulador UFSC} --- \emph{web}, sem instalação; construção gráfica.
    \item \textbf{AutomatoGraph} --- só AFDs; gera a gramática regular; simulação parcialmente visual.
    \item \textbf{FSM} (Morazán e Antunez, 2014) --- \textbf{textual}: autômatos como código Racket,
          testáveis e versionáveis; muitos formalismos; execução \textbf{só em texto}, visualização
          como trabalho futuro.
  \end{itemize}
\end{frame}

\begin{frame}{Análise comparativa\notexto{Seção 3.6}}

\section{Projeto da DSL}

\begin{frame}[fragile]{A linguagem: o mesmo autômato, em texto\notexto{Seção 6.1}}
  \begin{columns}[T]
    \begin{column}{0.60\textwidth}
\begin{lstlisting}[language=asl]
alfabeto { 'a', 'b' }

automato AFD exemplo_afd {
    estados { q0, q1, q2 }
    inicial q0
    finais { q2 }
    transicoes {
        q0 -> q1 com 'a'    q0 -> q0 com 'b'
        q1 -> q1 com 'a'    q1 -> q2 com 'b'
        q2 -> q2 com 'a'    q2 -> q2 com 'b'
    }
}

simular exemplo_afd com "ab"    // aceita
simular exemplo_afd com "bbb"   // rejeita
\end{lstlisting}
    \end{column}
    \begin{column}{0.40\textwidth}
      \small
      \begin{itemize}
        \item DSL \textbf{externa}, declarativa, em \textbf{português}.
        \item Cada elemento da quíntupla tem uma declaração própria.
        \item Arquivo \texttt{.asl}: alfabeto comum, autômatos e comandos \texttt{simular}.
        \item Texto puro: \textbf{versionável} com Git.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Gramática e decisões de projeto\notexto{Seções 6.2--6.4}}
  \footnotesize Trecho da gramática em EBNF (completa na Seção 6.2):
\begin{lstlisting}[basicstyle=\ttfamily\footnotesize,literate={->}{->}2]
declaracao_automato ::= "automato" ("AFD" | "AFN") ident "{"
                        "estados" lista "inicial" ident "finais" lista
                        "transicoes" "{" transicao* "}" "}"
transicao           ::= ident "->" ident "com" (char_literal | "epsilon" | "eps")
comando_simulacao   ::= "simular" ident "com" cadeia_literal
\end{lstlisting}
  \vspace{2pt}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}\small
        \item \textbf{Alfabeto único} por arquivo.
        \item Tipo \textbf{AFD/AFN explícito}: determinismo só é exigido do AFD.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}\small
        \item \textbf{Português} e \textbf{case-insensitive} (11 palavras reservadas).
        \item Transições $\varepsilon$ com \texttt{epsilon} ou \texttt{eps}.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Análise semântica e diagnósticos\notexto{Seções 7.4 e 7.7}}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \begin{enumerate}\small
        \item \textbf{Tabela de símbolos}: duplicatas; alfabeto vazio.
        \item \textbf{Referências}: estados, símbolos e autômatos não declarados.
        \item \textbf{Determinismo} (só AFD): sem $\varepsilon$ e uma transição por (estado, símbolo).
      \end{enumerate}
      \vspace{4pt}
      \small Erros \textbf{acumulados}: todos aparecem em uma só execução.
    \end{column}
    \begin{column}{0.56\textwidth}
      \centering
      \includegraphics[width=\textwidth]{img/erro_lexico_token_invalido.png}\\[4pt]
      \footnotesize Diagnóstico real da Autosim: arquivo, \textbf{linha e coluna},
      trecho realçado e descrição em português.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Simulador: um caminho, não uma nuvem de estados\notexto{Seção 7.5}}
  \begin{itemize}
    \item No AFN, acender \textbf{todos os estados ativos} ao mesmo tempo dificultava acompanhar
          a execução.
    \item A Autosim exibe um \textbf{caminho único}:
          \begin{itemize}
            \item cadeia aceita $\to$ um \textbf{caminho aceitador};

\begin{frame}[label=q-aprendizagem]{Como vocês sabem que a ferramenta ajuda o aluno?\notexto{Seções 4.4, 8.9 e 8.10}}
  \respostacurta{Não afirmamos isso. A validação é \textbf{funcional}: mostra que a retroalimentação
  é \textbf{produzida corretamente}. O efeito sobre a aprendizagem é uma \textbf{hipótese}, a ser
  verificada em estudo com turmas.}
  \begin{itemize}\small
    \item O recorte está declarado desde os objetivos (Cap. 1) e nas limitações (Seção 8.10).
    \item Um estudo controlado exigiria turmas comparáveis, pré e pós-teste e um semestre letivo:
          fora do prazo de um TCC.
    \item O questionário com 11 estudantes (Seção 8.9) mede \textbf{percepção}: indica
          aceitação, não aprendizagem.
    \item A motivação vem da literatura: falta de retroalimentação imediata (Morazán e Antunez, 2014).
  \end{itemize}
  \voltarperguntas
\end{frame}

\begin{frame}[label=q-contribuicao]{Qual é a contribuição original do trabalho?\notexto{Seções 3.6 e 4.1}}
  \respostacurta{Compiladores e simuladores existem; a contribuição é o \textbf{acoplamento}: o
  estudante escreve texto, um compilador aponta os erros no vocabulário dos autômatos, e o diagrama
  é gerado e animado a partir desse texto.}
  \begin{itemize}\small
    \item Entre as ferramentas analisadas, as gráficas não têm texto nem diagnóstico, e a textual
          (FSM) não visualiza.
    \item Wazlawick (2014): um novo artefato deve mostrar \textbf{em que difere} das soluções
          existentes (análise comparativa) e \textbf{verificar que funciona} (Cap. 8).
    \item Contribuições técnicas: a DSL em português com EBNF, o simulador de caminho único e a
          validação diferencial reprodutível.
  \end{itemize}
  \voltarperguntas
\end{frame}

\begin{frame}[label=q-jflap]{O que a Autosim faz que o JFLAP não faz?\notexto{Seções 3.2 e 3.6}}
  \respostacurta{A Autosim descreve o autômato em \textbf{texto}, versionável, e o valida com um
  \textbf{compilador} que localiza os erros. O JFLAP é muito mais abrangente; as duas são
  \textbf{complementares}.}

    \begin{column}{0.5\textwidth}
      \begin{block}{Autosim}
        \footnotesize
        \begin{itemize}
          \item Arquivo de texto: Git, \textit{diff}, reaproveitamento.
          \item Erros com linha, coluna e descrição.
          \item Determinismo verificado quando o aluno declara AFD.
          \item Binário nativo, sem instalar Java.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{JFLAP}
        \footnotesize
        \begin{itemize}
          \item Toda a hierarquia de Chomsky (pilha, Turing, gramáticas).
          \item Conversão AFN$\to$AFD, minimização, expressões regulares.
          \item Décadas de uso e maturidade.

\begin{frame}[label=q-fsm]{Por que uma DSL externa, e não uma biblioteca?\notexto{Seções 2.2.6 e 3.5}}
  \respostacurta{Uma DSL externa tem \textbf{só a sintaxe do domínio}: o estudante não precisa
  aprender a linguagem hospedeira, e o compilador fala em estados, transições e alfabeto.}
  \begin{itemize}\small
    \item Na FSM, o autômato é escrito na sintaxe de Racket; na Autosim, só com as construções
          do próprio domínio.
    \item A gramática restrita permite \textbf{verificações específicas}: estado não declarado,
          símbolo fora do alfabeto, determinismo do AFD.
    \item Custo: é preciso manter um compilador próprio, e a linguagem é menos expressiva
          (a FSM permite programar com os autômatos, compor, gerar testes aleatórios).
  \end{itemize}
  \voltarperguntas
\end{frame}

% ----- Linguagem e teoria ------------------------------------------------------------------------
\begin{frame}[fragile,label=q-ll1]{A gramática da linguagem é LL(1)?\notexto{Seções 6.2 e 7.3}}
  \respostacurta{A \textbf{linguagem} é LL(1). A EBNF, \textbf{como está escrita}, não é: na lista
  com vírgula final opcional, ao ver ``\texttt{,}'' é preciso olhar o próximo \textit{token}
  (outro item ou ``\texttt{\}}'') para decidir.}
  \begin{itemize}\small
    \item Fatorando à esquerda, a decisão passa a usar um único \textit{token}:
  \end{itemize}
\begin{lstlisting}[basicstyle=\ttfamily\footnotesize,literate={->}{->}2]
lista ::= "{" [ item resto ] "}"
resto ::= [ "," [ item resto ] ]
\end{lstlisting}
  \begin{itemize}\small
    \item O Chumsky não exige gramática LL(1): os combinadores tentam as alternativas em ordem.
  \end{itemize}
  \voltarperguntas
\end{frame}

\begin{frame}[label=q-parcial]{O AFD precisa ter todas as transições?\notexto{Seções 2.1.3 e 7.5}}
  \respostacurta{Não. Transições ausentes são permitidas: quando falta a transição, a simulação
  para e \textbf{rejeita}. Equivale a um \textbf{estado de rejeição implícito}, e a linguagem
  reconhecida não muda.}
  \begin{itemize}\small
    \item O estado implícito é não final e tem laço para todos os símbolos; acrescentá-lo
          torna $\delta$ total sem alterar a linguagem.
    \item É a convenção usual nos diagramas dos livros, e poupa o aluno de desenhar o estado de
          ``lixo''.
    \item O que a Autosim exige do AFD é o que \textbf{contradiz} o determinismo: transição
          $\varepsilon$ ou duas saídas para o mesmo (estado, símbolo).
    \item Coberto pelo teste ``rejeição por falta de transição'' (Seção 8.5).
  \end{itemize}
  \voltarperguntas
\end{frame}

\begin{frame}[label=q-afnafd]{E se um AFD for declarado como AFN?\notexto{Seção 6.4.2}}
  \respostacurta{É aceito: \textbf{todo AFD é um AFN} em que cada transição leva a um único
  estado. A classificação explícita serve para exigir determinismo quando o aluno
  \textbf{afirma} que o autômato é um AFD.}
  \begin{itemize}\small
    \item Declarar AFD é uma afirmação que o compilador verifica: se houver $\varepsilon$ ou duas
          saídas com o mesmo símbolo, o erro aponta as duas transições.
    \item Assim o erro conceitual ``achei que era determinístico'' vira um diagnóstico localizado.
  \end{itemize}
  \voltarperguntas
\end{frame}

\begin{frame}[label=q-foraalfabeto]{E se a cadeia tiver um símbolo fora do alfabeto?\notexto{Seção 7.4}}
  \respostacurta{Hoje a cadeia do \texttt{simular} \textbf{não é conferida} contra o alfabeto:
  o símbolo não tem transição, e a cadeia é \textbf{rejeitada}, sem mensagem específica.}
  \begin{itemize}\small
    \item O veredito está correto: uma cadeia com símbolo fora de $\Sigma$ não pertence a $L(M)
          \subseteq \Sigma^*$.
    \item Mas o diagnóstico seria mais útil: a verificação natural é ``símbolo `z' fora do
          alfabeto'' na análise semântica, como já ocorre nas \textbf{transições}.
    \item Reconhecemos como uma \textbf{melhoria} a fazer.
  \end{itemize}

\end{frame}

\begin{frame}[label=q-alfabeto]{Por que um único alfabeto por arquivo?\notexto{Seção 6.4.1}}

\begin{frame}[label=q-rust]{Por que usaram Rust?\notexto{Seção 5.4}}
  \respostacurta{Por quatro requisitos: \textbf{binário nativo} sem ambiente de execução,
  \textbf{enumerações com \texttt{match} exaustivo} para AST e erros, \textbf{bibliotecas de

\begin{frame}[label=q-caminho]{Por que mostrar um caminho, e não todos os estados?\notexto{Seção 7.5}}
  \respostacurta{Com vários estados acesos ao mesmo tempo, a execução ficava \textbf{difícil de
  acompanhar}. Um caminho único é legível, e a escolha garante que ele seja \textbf{aceitador}
  sempre que a cadeia pertence à linguagem.}
  \begin{itemize}\small
    \item Busca em largura sobre (estado, símbolos consumidos): vence quem consumiu mais e está
          mais perto de um estado final.
    \item O veredito é sempre o da definição formal: a busca explora todas as configurações.
    \item \textbf{Ciclos de $\varepsilon$} não travam: cada par é visitado uma vez.
    \item Detalhes: \hyperlink{res-caminho}{\textcolor{acento}{como o caminho é escolhido}}.
  \end{itemize}
  \voltarperguntas
\end{frame}

\begin{frame}[label=q-escala]{A ferramenta funciona com autômatos grandes?\notexto{Seções 7.5, 7.6 e 8.7}}

  custo polinomial. O limite está no \textbf{desenho}: o arranjo circular perde legibilidade com
  muitos estados.}
  \begin{itemize}\small
    \item Na validação, 8.191 cadeias de um AFN foram simuladas em frações de segundo.
    \item O diagrama foi testado com um AFD de 14 estados: o raio cresce com o número de estados,
          e há \textit{pan} e \textit{zoom}.
    \item Para autômatos de dezenas de estados, um algoritmo de desenho de grafos reduziria
          cruzamentos de arestas: possível melhoria.
  \end{itemize}
  \voltarperguntas
\end{frame}

\begin{frame}[label=res-chomsky]{Hierarquia de Chomsky\notexto{Seção 2.1.2}}
  \begin{center}
    \small
    \begin{tabular}{@{}clll@{}}
      \toprule
      \textbf{Tipo} & \textbf{Linguagens} & \textbf{Regras} & \textbf{Modelo computacional} \\
      \midrule
      0 & Recursivamente enumeráveis & $\alpha \to \beta$ & Máquina de Turing \\
      1 & Sensíveis ao contexto & $\alpha A \beta \to \alpha \gamma \beta$ & Autômato linearmente limitado \\
      2 & Livres de contexto & $A \to \gamma$ & Autômato com pilha \\
      \rowcolor{acento!12}
      3 & \textbf{Regulares} & $A \to aB$ ou $A \to a$ & \textbf{Autômato finito} \\
      \bottomrule
    \end{tabular}
  \end{center}
  \vspace{6pt}
  \begin{itemize}\small
    \item A Autosim cobre o \textbf{tipo 3}: AFDs e AFNs reconhecem exatamente as linguagens regulares.
    \item Exemplo fora do alcance: $\{a^n b^n \mid n \ge 0\}$ é livre de contexto e exige pilha.
    \item Trabalhos futuros: subir a hierarquia com \textbf{autômatos com pilha} (tipo 2) e
          \textbf{máquinas de Turing} (tipo 0), com fita e cabeçote animados.
  \end{itemize}

\begin{frame}[label=res-subconjuntos]{Construção de subconjuntos (AFN $\to$ AFD)\notexto{Seção 2.1.5}}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \small
      Cada estado do AFD é um \textbf{conjunto} de estados do AFN:
      \begin{itemize}\footnotesize
        \item $q_{0,D} = \{q_{0,N}\}$;
        \item $\delta_D(S, a) = \bigcup_{p \in S} \delta_N(p, a)$;
        \item $S$ é final se $S \cap F_N \neq \varnothing$.
      \end{itemize}
      \vspace{2pt}
      Só se constroem os subconjuntos \textbf{alcançáveis}: das $2^3 = 8$ possibilidades,
      o exemplo usa 3.\\[4pt]
      \footnotesize Pior caso: $2^n$ estados. AFD e AFN reconhecem a mesma classe de linguagens.
    \end{column}
    \begin{column}{0.54\textwidth}

\begin{frame}[label=res-ownership]{\textit{Ownership}, \textit{borrowing} e Cargo\notexto{Seções 5.2--5.3}}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Memória sem coletor de lixo}
        \small
        \begin{itemize}
          \item \textbf{\textit{Ownership}}: cada valor tem um único dono e é liberado quando
                ele sai de escopo.
          \item \textbf{\textit{Borrowing}}: várias referências \texttt{\&T} \textbf{ou} uma
                \texttt{\&mut T}, verificado na compilação.
          \item Na Autosim: as fases leem a AST por referência, sem cópias.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Tipos e ferramentas}
        \small
        \begin{itemize}
          \item \texttt{enum} + \texttt{match} \textbf{exaustivo}: todo \textit{token}, nó da
                AST e erro é tratado, ou o código não compila.
          \item \textbf{Cargo}: dependências (\texttt{Cargo.toml}, \texttt{crates.io}),
                compilação e testes (\texttt{cargo test}) em uma só ferramenta.
        \end{itemize}
      \end{block}

\begin{frame}[label=res-caminho]{Como o caminho do AFN é escolhido\notexto{Seção 7.5.2}}

    \item \textbf{Busca em largura} sobre os pares (estado, símbolos consumidos), seguindo
          transições $\varepsilon$ (não avançam) e com o próximo símbolo (avançam uma
          posição). Cada par é visitado uma vez: ciclos de $\varepsilon$ não travam a busca.
    \item Entre os pares alcançados, elege-se o melhor por:
          \begin{enumerate}\footnotesize
            \item \textbf{maior} número de símbolos consumidos;
            \item \textbf{menor} distância até um estado final.
          \end{enumerate}
    \item A distância vem de uma \textbf{segunda busca em largura}, no grafo invertido, a
          partir de todos os estados finais (calculada uma vez, antes).
    \item O caminho é reconstruído pelos apontadores da busca e é o \textbf{mais curto} até
          aquele par.
  \end{enumerate}
  \vspace{4pt}
  \begin{block}{Por que funciona}
    \footnotesize Se a cadeia pertence à linguagem, há um par que consumiu tudo e está em um estado
    final (distância 0): ele vence, e o caminho exibido é \textbf{aceitador}. Se não pertence,
    vence o caminho que chega mais longe e termina mais perto de um final. O veredito é sempre o
    da definição formal.
  \end{block}

\end{frame}

\begin{frame}[label=res-dificuldades]{Dificuldades encontradas\notexto{Cap. 9}}
  \begin{enumerate}\small
    \item \textbf{AFN com vários estados acesos}: a primeira versão propagava o conjunto de estados
          e a execução ficava difícil de seguir. $\to$ Escolha prévia de um \textbf{caminho
          único} (Seção 7.5.2).
    \item \textbf{Autômatos grandes}: estados se sobrepunham ou saíam da janela. $\to$ Raio do
          arranjo cresce com o número de estados, e \textbf{\textit{pan} e \textit{zoom}}.
    \item \textbf{Localização dos erros}: a AST guarda índices de \textit{tokens}, e acentos em
          comentários deslocavam a coluna (\textit{bytes} lidos como caracteres). $\to$
          Conversão de volta ao texto e \textbf{\textit{bytes} em toda a cadeia} de diagnóstico.
    \item \textbf{Coerência entre texto, código e teoria}: a gramática e um teste aceitavam
          \texttt{alfabeto \{ \}}, mas a definição exige alfabeto não vazio. $\to$
          \textbf{Verificação semântica} específica.
  \end{enumerate}
