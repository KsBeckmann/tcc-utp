\section{Introdução}

\begin{frame}{Contexto e problema}
  \begin{itemize}
    \item Autômatos finitos são um pilar da Ciência da Computação, mas a disciplina é
          \textbf{reconhecidamente difícil}.
    \item Morazán e Antunez (2014): o estudante projeta autômatos \textbf{sem poder experimentar}
          nem receber \textbf{retroalimentação imediata}, ao contrário da programação.
    \item As ferramentas mais difundidas (\textbf{JFLAP}) são \textbf{só visuais}:
          sem descrição textual, sem versionamento, \textbf{sem diagnóstico} de erros.
    \item A alternativa textual (\textbf{FSM}, em Racket) descreve autômatos como código,
          mas \textbf{não os desenha nem anima}.
  \end{itemize}
  \vspace{4pt}
  \begin{block}{Lacuna}
    Unir a \textbf{descrição textual}, o \textbf{diagnóstico de um compilador} e a
    \textbf{simulação visual} em uma única ferramenta.
  \end{block}
\end{frame}

\begin{frame}{Pergunta de pesquisa e proposta}
  \begin{block}{Pergunta de pesquisa}
    Como uma linguagem textual de domínio específico, processada por um compilador com
    diagnósticos localizados e acoplada a uma simulação visual, pode oferecer ao estudante
    \textbf{retroalimentação imediata} na construção e na execução de autômatos finitos?
  \end{block}
  \vspace{6pt}
  \centering
  \begin{tikzpicture}[
      node distance=9mm,
      parte/.style={draw=acento,fill=acento!10,rounded corners,align=center,
                    minimum height=13mm,text width=33mm,inner sep=4pt,font=\small},
      seta/.style={-{Stealth[length=2.5mm]},very thick,acento}]
    \node[parte] (d) {\textbf{DSL textual}\\em português};
    \node[parte,right=of d] (c) {\textbf{Compilador}\\diagnósticos com\\linha e coluna};
    \node[parte,right=of c] (i) {\textbf{Interface gráfica}\\diagrama e simulação\\passo a passo};
    \draw[seta] (d)--(c); \draw[seta] (c)--(i);
    \node[below=3mm of c,font=\small] {\textbf{Autosim}: o estudante escreve o texto; o diagrama é gerado e animado automaticamente};
  \end{tikzpicture}
\end{frame}

\begin{frame}{Objetivos}
  \textbf{Objetivo geral:} desenvolver uma ferramenta educacional que integra a descrição textual
  de autômatos, sua compilação pedagógica e sua simulação visual em um único ambiente.
  \vspace{6pt}
  \begin{block}{Objetivos específicos}
    \begin{enumerate}
      \item Projetar uma \textbf{DSL} textual para AFDs e AFNs, com gramática em EBNF;
      \item Implementar um \textbf{compilador pedagógico} com diagnósticos localizados;
      \item Desenvolver um \textbf{simulador} de cadeias sobre os autômatos descritos;
      \item Construir uma \textbf{interface gráfica} que desenhe e anime o autômato;
      \item \textbf{Validar} a ferramenta com testes, autômatos clássicos e cenários de erro.
    \end{enumerate}
  \end{block}
  \vspace{2pt}
  {\footnotesize Validação \textbf{funcional}; a eficácia pedagógica com estudantes é trabalho futuro.}
\end{frame}

\section{Fundamentação Teórica}

\begin{frame}{Autômatos finitos}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      Um \textbf{AFD} é uma quíntupla $M = (Q, \Sigma, \delta, q_0, F)$:
      \begin{itemize}\small
        \item $Q$: estados \quad $\Sigma$: alfabeto
        \item $\delta: Q \times \Sigma \to Q$: transições
        \item $q_0$: inicial \quad $F \subseteq Q$: finais
      \end{itemize}
      \vspace{4pt}
      No \textbf{AFN}, $\delta: Q \times (\Sigma \cup \{\varepsilon\}) \to \mathcal{P}(Q)$:
      vários destinos e transições vazias $\varepsilon$.
      \vspace{4pt}

      AFD e AFN são \textbf{equivalentes} (construção de subconjuntos) e reconhecem as
      \textbf{linguagens regulares}.
    \end{column}
    \begin{column}{0.53\textwidth}
      \centering
      \vspace{14pt}
      \begin{tikzpicture}[shorten >=1pt,node distance=2.3cm,on grid,>={Stealth[]},
          every state/.style={draw=acento,thick,fill=acento!8,minimum size=9mm}]
        \node[state,initial,initial text={}] (q0) {$q_0$};
        \node[state] (q1) [right=of q0] {$q_1$};
        \node[state,accepting] (q2) [right=of q1] {$q_2$};
        \path[->]
          (q0) edge[loop above] node{$b$} (q0)
          (q0) edge node[above]{$a$} (q1)
          (q1) edge[loop above] node{$a$} (q1)
          (q1) edge node[above]{$b$} (q2)
          (q2) edge[loop above] node{$a,b$} (q2);
      \end{tikzpicture}
      \vspace{8pt}

      \small Cadeias que \textbf{contêm ``$ab$''}\\[3pt]
      \textcolor{aceita}{\textbf{aceita:}} \texttt{ab}, \texttt{aab}, \texttt{bab}\\
      \textcolor{rejeita}{\textbf{rejeita:}} $\varepsilon$, \texttt{ba}, \texttt{bbb}\\[3pt]
      \footnotesize\emph{Reaparece na DSL e na demonstração.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Compiladores e linguagens de domínio específico}
  \begin{center}
    \begin{tikzpicture}[
        node distance=6mm and 10mm,

      \draw[seta] (a)--(b); \draw[seta] (b)--(c);

      \node[below=1mm of b,font=\footnotesize\itshape] {\emph{tokens} $\to$ AST};
      \node[below=1mm of c,font=\footnotesize\itshape,text width=30mm,align=center] {referências e consistência};
    \end{tikzpicture}
  \end{center}
  \vspace{2pt}
  \begin{block}{DSL (Fowler, 2010)}
    Linguagem de \textbf{expressividade limitada}, voltada a um \textbf{domínio específico}.
    \begin{itemize}
      \item \textbf{Interna:} embutida em uma linguagem hospedeira (ex.: FSM, em Racket).
      \item \textbf{Externa:} sintaxe própria e \textbf{compilador dedicado} --- é o caso da \textbf{Autosim}.
    \end{itemize}
  \end{block}
\end{frame}

\section{Trabalhos Relacionados}

\begin{frame}{Panorama: quatro ferramentas gráficas e uma textual}
  \begin{itemize}
    \item \textbf{JFLAP} --- referência mundial; cobre toda a hierarquia de Chomsky;
          construção \textbf{só gráfica}, sem descrição textual nem diagnóstico.
    \item \textbf{GAM} (UFLA) --- \emph{desktop}; limitado a 10 estados e 3 símbolos.
    \item \textbf{Simulador UFSC} --- \emph{web}, sem instalação; construção gráfica.
    \item \textbf{AutomatoGraph} --- só AFDs; gera a gramática regular; simulação parcialmente visual.
    \item \textbf{FSM} (Morazán e Antunez, 2014) --- \textbf{textual}: autômatos como código Racket,
          testáveis e versionáveis; muitos formalismos; execução \textbf{só em texto}, visualização
          como trabalho futuro.
  \end{itemize}
\end{frame}

\begin{frame}{Análise comparativa}
  \begin{center}
    \small
    \setlength{\tabcolsep}{5pt}

\end{frame}

\section{Projeto da DSL}

\begin{frame}[fragile]{A linguagem: o mesmo autômato, em texto}
  \begin{columns}[T]
    \begin{column}{0.60\textwidth}
\begin{lstlisting}[language=asl]
alfabeto { 'a', 'b' }

automato AFD exemplo_afd {
    estados { q0, q1, q2 }
    inicial q0
    finais { q2 }
    transicoes {
        q0 -> q1 com 'a'    q0 -> q0 com 'b'
        q1 -> q1 com 'a'    q1 -> q2 com 'b'
        q2 -> q2 com 'a'    q2 -> q2 com 'b'
    }
}

simular exemplo_afd com "ab"    // aceita
simular exemplo_afd com "bbb"   // rejeita
\end{lstlisting}
    \end{column}
    \begin{column}{0.40\textwidth}
      \small
      \begin{itemize}
        \item DSL \textbf{externa}, declarativa, em \textbf{português}.
        \item Cada elemento da quíntupla tem uma declaração própria.
        \item Arquivo \texttt{.asl}: alfabeto comum, autômatos e comandos \texttt{simular}.
        \item Texto puro: \textbf{versionável} com Git.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Gramática e decisões de projeto}
  \footnotesize Trecho da gramática em EBNF (completa na Seção 6.2):
\begin{lstlisting}[basicstyle=\ttfamily\footnotesize,literate={->}{->}2]
declaracao_automato ::= "automato" ("AFD" | "AFN") ident "{"
                        "estados" lista "inicial" ident "finais" lista
                        "transicoes" "{" transicao* "}" "}"
transicao           ::= ident "->" ident "com" (char_literal | "epsilon" | "eps")
comando_simulacao   ::= "simular" ident "com" cadeia_literal
\end{lstlisting}
  \vspace{2pt}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}\small
        \item \textbf{Alfabeto único} por arquivo.
        \item Tipo \textbf{AFD/AFN explícito}: determinismo só é exigido do AFD.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}\small
        \item \textbf{Português} e \textbf{case-insensitive} (11 palavras reservadas).
        \item Transições $\varepsilon$ com \texttt{epsilon} ou \texttt{eps}.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Análise semântica e diagnósticos}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \begin{enumerate}\small
        \item \textbf{Tabela de símbolos}: duplicatas; alfabeto vazio.
        \item \textbf{Referências}: estados, símbolos e autômatos não declarados.
        \item \textbf{Determinismo} (só AFD): sem $\varepsilon$ e uma transição por (estado, símbolo).
      \end{enumerate}
      \vspace{4pt}
      \small Erros \textbf{acumulados}: todos aparecem em uma só execução.
    \end{column}
    \begin{column}{0.56\textwidth}
      \centering
      \includegraphics[width=\textwidth]{img/erro_lexico_token_invalido.png}\\[4pt]
      \footnotesize Diagnóstico real da Autosim: arquivo, \textbf{linha e coluna},
      trecho realçado e descrição em português.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Simulador: um caminho, não uma nuvem de estados}
  \begin{itemize}
    \item No AFN, acender \textbf{todos os estados ativos} ao mesmo tempo dificultava acompanhar
          a execução.
    \item A Autosim exibe um \textbf{caminho único}:
          \begin{itemize}
            \item cadeia aceita $\to$ um \textbf{caminho aceitador};
            \item cadeia rejeitada $\to$ o caminho que \textbf{mais se aproximou} da aceitação.
          \end{itemize}
    \item Escolhido por \textbf{busca em largura} sobre (estado, símbolos consumidos): o veredito
          continua sendo o da definição formal.
    \item \textbf{Histórico completo}: avançar e voltar passo a passo.
  \end{itemize}
\end{frame}

\section{Conclusão}

\begin{frame}{Contribuições}
  A pergunta de pesquisa foi respondida no \textbf{plano funcional}: a Autosim entrega
  retroalimentação imediata na construção e na execução de autômatos.
  \begin{itemize}
    \item \textbf{DSL} declarativa em português para AFDs e AFNs com $\varepsilon$, em EBNF.
    \item \textbf{Compilador pedagógico} com diagnósticos localizados.
    \item \textbf{Simulador} de caminho único, com histórico nos dois sentidos.
    \item \textbf{Interface gráfica} que desenha e anima o autômato a partir do texto.
    \item Validação: \textbf{51 testes}, \textbf{57.368 simulações} sem divergência e cenários de erro.
    \item Distribuída para Linux, Windows e macOS (versão 1.0.4), com manual de uso.
  \end{itemize}
\end{frame}

\begin{frame}{Limitações e trabalhos futuros}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Limitações}
        \begin{itemize}\small
          \item Alfabeto único por arquivo.
          \item Sem modo de linha de comando.
          \item Um arquivo por execução.
          \item Sem editor embutido com realce.
          \item Sem conversão AFN$\to$AFD nem minimização.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Trabalhos futuros}
        \begin{itemize}\small
          \item Editor com realce na interface.
          \item Conversões AFN$\to$AFD, minimização e expressões regulares.
          \item Autômatos com pilha e \textbf{máquinas de Turing}.
          \item \textbf{Estudos controlados com estudantes}.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
